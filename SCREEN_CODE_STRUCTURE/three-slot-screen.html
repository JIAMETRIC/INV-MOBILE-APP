Three-Slot Screen (3SS)

Three slots: Header, Main, Footer. Inside Main, rows and columns. Every region is a named function carrying its layout ID (S2.3a, S4.B.3). The name says the shape; the IDs say where.

Also fits as a Claude Code skill: three-slot-screen. A skill is a procedure the assistant runs; the design spec stays what things look like.

1 · The scaffold

src/ui/Screen.tsx · three slots, insets, background, flex 1 written once
// src/ui/Screen.tsx — Header and Footer keep their own height; Main fills the rest.
import React from 'react';
import { View } from 'react-native';
import { useTheme } from '../shell/providers';
import { useNavBottom } from './navMetrics';

export function Screen({ header, footer, children }: {
  header?: React.ReactNode; footer?: React.ReactNode; children: React.ReactNode;
}) {
  const theme = useTheme();
  const navBottom = useNavBottom(); // max(gesture inset, 20)
  return (
    <View style={{ flex: 1, backgroundColor: theme.bg }}>
      {header}
      <View style={{ flex: 1, minHeight: 0 }}>{children}</View>
      {footer ? <View style={{ paddingBottom: navBottom }}>{footer}</View> : null}
    </View>
  );
}

const S5 = {
  searchGap: 10,                      // S5.1: search field | export button
  fabClearance: touch.button + layout.gap, // S5.3: last row clears the FAB
  editorWidth: '38%',                 // S5.5: tablet editor pane
} as const;

// All state and data. No JSX.
function useProductList() {
  const [rows, setRows] = useState<ProductRow[]>([]);
  const [search, setSearch] = useState('');
  useEffect(() => { void listProducts(search).then(setRows); }, [search]);
  return { rows, search, setSearch };
}
type ProductList = ReturnType<typeof useProductList>;

// The switch: phone = Header + Main + Footer; tablet = Header + [Main | Editor].
export function ProductsScreen() {
  const theme = useTheme();
  const bp = useBreakpoint();
  const list = useProductList();
  return (
    <Screen
      header={<Header list={list} theme={theme} />}
      footer={bp.isPhone ? <Footer theme={theme} /> : null}
    >
      <Main list={list} theme={theme} />
    </Screen>
  );
}

// S5.0 — title, "N items", export button.
function Header({ list, theme }: { list: ProductList; theme: Theme }) {
  return <TopAppBar title="Products" subtitle={`${list.rows.length} items`} large theme={theme} />;
}

// S5.1–S5.3 — the column under the header: search row, then the list card.
function Main({ list, theme }: { list: ProductList; theme: Theme }) {
  return (
    <View style={{ flex: 1, paddingHorizontal: layout.side, gap: layout.gap }}>
      <SearchRow list={list} theme={theme} />
      <ProductListCard list={list} theme={theme} />
    </View>
  );
}

// S5.4 — phone only.
function Footer({ theme }: { theme: Theme }) {
  const nav = useNav();
  return <FAB icon="plus" label="Add product" onPress={() => nav.openProduct()} theme={theme} />;
}

4 · A list with a memoised row

one FlatList in Main · the row is its own component · last row clears the footer
// S2.5 — the card hugs its rows and scrolls when full. Never wrap it in a ScrollView.
function InvoiceListCard({ list, theme }: { list: InvoiceList; theme: Theme }) {
  const navInset = useNavInset();
  const { rows } = list;
  return (
    <Card theme={theme} list style={{ flexShrink: 1, marginBottom: navInset, overflow: 'hidden' }}>
      <FlatList
        data={rows}
        keyExtractor={r => String(r.id)}
        style={{ flexGrow: 0 }}
        contentContainerStyle={{ paddingBottom: rows.length > S2.longList ? S2.fabClearance : 0 }}
        renderItem={({ item, index }) => (
          <InvoiceRow item={item} alt={index % 2 === 1} last={index === rows.length - 1} theme={theme} />
        )}
      />
    </Card>
  );
}

// One row: avatar · party · "no · date" · ₹ total over the status pill.
const InvoiceRow = React.memo(function InvoiceRow({ item, alt, last, theme }: {
  item: Invoice; alt: boolean; last: boolean; theme: Theme;
}) {
  return (
    <ListItem theme={theme} flush leadingText={initials(item.party_name)} leadingAlt={alt} divider={!last}
      title={item.party_name} subtitle={`${item.invoice_no} · ${formatDateDisplay(item.invoice_date)}`}
      trailing={<MoneyText paiseText={`₹${formatPaise(item.total_paise)}`} theme={theme} small />} />
  );
});

5 · A footer action bar

S4.L1.3 · running total + the step's primary action · rises with the keyboard
// Footer: a floating glass bar (bottom-nav recipe). Left = TOTAL, right = the action.
function Footer({ total, step, onNext, actions, theme }: {
  total: number; step: number; onNext: () => void; actions: React.ReactNode; theme: Theme;
}) {
  return (
    <View style={{
      marginHorizontal: layout.side, marginTop: space[2],
      paddingStart: 20, paddingEnd: space[2], paddingVertical: space[2],
      borderRadius: radius.card, backgroundColor: theme.navSurface,
      borderWidth: 1, borderColor: theme.navBorder, boxShadow: shadow.secondary,
      flexDirection: 'row', alignItems: 'center', gap: space[2],
    }}>
      <View style={{ flex: 1 }}>
        <Text style={[type.overline, { color: theme.textMuted }]}>Total</Text>
        <Text style={[type.figure, { color: theme.text }]} numberOfLines={1}>₹{formatPaise(total)}</Text>
      </View>
      {step < 2 ? <Button label="Next" icon="arrow-right" theme={theme} onPress={onNext} /> : actions}
    </View>
  );
}
