Every screen is a column of three slots on <Screen>. Inside Main, rows stack top to bottom and a row may split into columns with <Columns>. Each slot and each row is a named function in the screen file, so an edit means opening the region by name. All six screens of the app are on it; every screen was checked at eight phone and tablet sizes.

Three-Slot Screen (3SS)

Three slots: Header, Main, Footer. Inside Main, rows and columns. Every region is a named function carrying its layout ID (S2.3a, S4.B.3). The name says the shape; the IDs say where.

Also fits as a Claude Code skill: three-slot-screen. A skill is a procedure the assistant runs; the design spec stays what things look like.

The tree order is also the screen-reader order: Header, then rows left to right, then Footer. A hosted screen (S3 Invoice detail, shown under the shell's back bar or in S2's pane) is an embeddable Main: no <Screen>, no header or footer of its own.

1 · The scaffold

src/ui/Screen.tsx (C16) · slots, insets, background and the nav / FAB clearance written once
// src/ui/Screen.tsx (C16) — three slots, a fab slot, the insets; the ONE place that knows the clearances.
import React, { createContext, useContext } from 'react';
import { View } from 'react-native';
import { useTheme } from '../shell/providers';
import { layout, space, touch } from './tokens';
import { useNavBottom, useNavInset } from './navMetrics';

const ScreenContext = createContext({ fab: false, overlay: false });

export function Screen({ header, below, footer, fab, overlay = false, children }: {
  header?: React.ReactNode;   // fixed; a TopAppBar pads the status inset itself
  below?: React.ReactNode;    // optional fixed row under the header (a stepper)
  footer?: React.ReactNode;   // phones only; in the flex column, rises with the keyboard
  fab?: React.ReactNode;      // floats over Main
  overlay?: boolean;          // a full-screen editor or detail: no bottom nav under it
  children: React.ReactNode;  // Main: exactly one scroller, or a row of panes
}) {
  const theme = useTheme();
  const navBottom = useNavBottom(); // max(gesture inset, 20)
  return (
    <ScreenContext.Provider value={{ fab: !!fab, overlay }}>
      <View style={{ flex: 1, backgroundColor: theme.bg }}>
        {header}
        {below}
        <View style={{ flex: 1, minHeight: 0 }}>{children}</View>
        {footer ? <View style={{ paddingBottom: navBottom }}>{footer}</View> : null}
        {fab}
      </View>
    </ScreenContext.Provider>
  );
}

// The FAB's height (52) plus a block gap, only while a fab floats.
export function useFabClearance(): number {
  const { fab } = useContext(ScreenContext);
  return fab ? touch.button + layout.gap : 0;
}

// Bottom padding for Main's scroller: the floating nav (+ the fab), or a gap in an overlay.
export function useMainInset(): number {
  const { overlay } = useContext(ScreenContext);
  return (overlay ? space[6] : useNavInset()) + useFabClearance();
}
// A footer in the column takes its own space: nothing to measure.

  tableMinWidth: 520,     // S5.3: narrower than this, the table becomes card rows
  editorWidth: '38%',     // S5.5: tablet editor pane

} as const;

// All state and data. No JSX. Row handlers are useCallback so memo rows stay cheap.
function useProducts() {
  const [rows, setRows] = useState<ProductRow[]>([]);
  const [search, setSearch] = useState('');
  useEffect(() => { void listProducts(search).then(setRows); }, [search]);
  const openRow = useCallback((id: number) => setEditing(id), []);
  return { rows, search, setSearch, openRow };
}
type Products = ReturnType<typeof useProducts>;

// The switch: phone = Header + Main + a floating FAB; lg+ = Header + [list | editor pane].
export function ProductsScreen() {
  const theme = useTheme();
  const bp = useBreakpoint();
  const p = useProducts();
  return (
    <Screen
      header={<Header p={p} theme={theme} />}
      fab={bp.isPhone ? <FAB icon="plus" label="Add product" onPress={p.openNew} theme={theme} /> : null}
    >
      {bp.panes === 1
        ? <ListPane p={p} theme={theme} />
        : <View style={{ flex: 1, flexDirection: 'row' }}>
            <ListPane p={p} theme={theme} />
            <EditorPane p={p} theme={theme} />   {/* width S5.editorWidth, flexGrow 0, flexShrink 0 */}
          </View>}
    </Screen>
  );
}

// S5.0 — title, "N items", export button.
function Header({ p, theme }: { p: Products; theme: Theme }) {
  return <TopAppBar title="Products" subtitle={`${p.rows.length} items`} large theme={theme} />;
}

// S5.1–S5.3 — the column under the header: search row, then the list card (the one scroller).

}

3 · A row with two columns

// S4.A row 3 — City | Pincode: one row, two equal columns. `kind="input"` keeps each cell
// ≥ 128 dp; narrower than that the row stacks. The decision uses the row's own measured width,
// so it is right inside a tablet pane too. No per-row breakpoint code.
function CityPincodeRow({ form, patch, theme }: { form: InvoiceForm; patch: Patch; theme: Theme }) {
  return (
    <Columns kind="input">
      <Input label="City" value={form.party.party_city}
        onChange={v => patch({ party: { ...form.party, party_city: v } })} theme={theme} />
      <Input label="Pincode" value={form.party.party_pincode} keyboardType="numeric" maxLength={6}
        onChange={v => patch({ party: { ...form.party, party_pincode: v } })} theme={theme} />
    </Columns>
  );
}
// Unequal cells: weights={[1, 1.4]}. Force a stack on phones: stackAt="sm".
// A 4th cell: add `wrap` and it goes to the next line on a phone (2 + 2) and
// comes back up on a tablet. Icon buttons: kind="touch" (44). Buttons: kind="button" (96).

4 · A list with a memoised row

one FlatList in Main · memo row with a useCallback handler · useFabClearance · filters in the list header on short screens
// S2.5 — the card hugs its rows and scrolls when full. Never wrap it in a ScrollView.
// It pads for the FAB with useFabClearance() (0 when no fab floats): no hand-written 52 + 16.
function InvoiceListCard({ list, theme }: { list: InvoiceList; theme: Theme }) {
  const navInset = useNavInset();
  const fabClearance = useFabClearance();
  const { rows, openRow } = list; // openRow is a useCallback in the hook
  return (
    <Card theme={theme} list style={{ flexShrink: 1, marginBottom: navInset, overflow: 'hidden' }}>
      <FlatList
        data={rows}
        keyExtractor={r => String(r.id)}
        style={{ flexGrow: 0 }}
        ListHeaderComponent={shortScreen ? <FilterRows list={list} theme={theme} /> : null}
        contentContainerStyle={{ paddingBottom: rows.length > S2.longList ? fabClearance : 0 }}
        renderItem={({ item, index }) => (
          <InvoiceRow item={item} alt={index % 2 === 1} last={index === rows.length - 1}
            onPress={openRow} theme={theme} />
        )}
      />
    </Card>
  );
}

// One row: avatar · party · "no · date" · ₹ total over the status pill. Memoised: a keystroke
// in the search box re-renders the list, not every row.
const InvoiceRow = React.memo(function InvoiceRow({ item, alt, last, onPress, theme }: {
  item: Invoice; alt: boolean; last: boolean; onPress: (id: number) => void; theme: Theme;
}) {
  return (
    <ListItem theme={theme} flush leadingText={initials(item.party_name)} leadingAlt={alt} divider={!last}
      title={item.party_name} subtitle={`${item.invoice_no} · ${formatDateDisplay(item.invoice_date)}`}
      trailing={<MoneyText paiseText={`₹${formatPaise(item.total_paise)}`} theme={theme} small />}
      onPress={() => onPress(item.id)} />
  );
});

5 · A footer action bar

S4.L1.3 · running total + the step's primary action · rises with the keyboard
// Footer: a floating glass bar (bottom-nav recipe). Left = TOTAL, right = the action.
function Footer({ total, step, onNext, actions, theme }: {
  total: number; step: number; onNext: () => void; actions: React.ReactNode; theme: Theme;
}) {
  return (
    <View style={{
      marginHorizontal: layout.side, marginTop: space[2],
      paddingStart: 20, paddingEnd: space[2], paddingVertical: space[2],
      borderRadius: radius.card, backgroundColor: theme.navSurface,
      borderWidth: 1, borderColor: theme.navBorder, boxShadow: shadow.secondary,
      flexDirection: 'row', alignItems: 'center', gap: space[2],
    }}>
      <View style={{ flex: 1 }}>
        <Text style={[type.overline, { color: theme.textMuted }]}>Total</Text>
        <Text style={[type.figure, { color: theme.text }]} numberOfLines={1}>₹{formatPaise(total)}</Text>
      </View>
      {step < 2 ? <Button label="Next" icon="arrow-right" theme={theme} onPress={onNext} /> : actions}
    </View>
  );
}
